
services:
  # --- Backend API (προαιρετικό, placeholder) ---
  backend:
    build: ./backend
    container_name: myai_backend
    restart: always
    env_file:
      - .env
      - ./backend/.env
    expose:
      - "8000"
    environment:
      - POSTGRES_HOST=db
      - POSTGRES_PORT=5432
      - REDIS_URL=redis://myai_redis:6379/0
      - QDRANT_URL=http://myai_qdrant:6333
      - OPENAI_API_KEY=${OPENAI_API_KEY}
    networks:
      - myai_network
    depends_on:
      - db
      - redis
      - qdrant
    command: uvicorn main:app --host 0.0.0.0 --port 8000

  # --- PostgreSQL Database ---
  db:
    image: postgres:16
    container_name: myai_postgres
    restart: always
    environment:
      - POSTGRES_USER=${POSTGRES_USER}
      - POSTGRES_PASSWORD=${POSTGRES_PASSWORD}
      - POSTGRES_DB=${POSTGRES_DB}
    networks:
      - myai_network
    volumes:
      - pg_data:/var/lib/postgresql/data

  # --- Redis ---
  redis:
    image: redis:7
    container_name: myai_redis
    restart: always
    expose:
      - "6379"
    networks:
      - myai_network

  # --- Qdrant Vector DB ---
  qdrant:
    image: qdrant/qdrant:latest
    container_name: myai_qdrant
    restart: always
    ports:
      - "6333:6333"
      - "6334:6334"
    environment:
      - QDRANT__SERVICE__ENABLE_UI=true
    volumes:
      - qdrant_data:/qdrant/storage
    networks:
      - myai_network

  # --- Nginx Proxy ---
  nginx:
    image: nginx:alpine
    container_name: myai_nginx
    restart: always
    depends_on:
      - backend
    ports:
      - "80:80"
      - "443:443"
    volumes:
     # - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
      - ./nginx/nginx.fixed.conf:/etc/nginx/nginx.conf:ro
     # - ./certbot/www:/usr/share/nginx/html
     # - ./certbot/conf:/etc/letsencrypt
      - ./frontend/build:/usr/share/nginx/html:ro
     # SECOND UI
      - ./frontend-second/build:/usr/share/nginx/second
     # TEST UI
      - ./test/build:/usr/share/nginx/test
     # Edo serviretai to periexomeno (HTML, react, test pages)
     # - ./certbot/www:/usr/share/nginx/html
      - ./certbot/www:/usr/share/nginx/certbot:ro
     # edo kanoume save ta SSL
      - certbot_certs:/etc/letsencrypt

    environment:
      - DOMAIN=${DOMAIN}
    networks:
      - myai_network

  adminer:
    image: adminer
    container_name: adminer
    restart: always
    ports:
      - "8082:8080"
    depends_on:
      - db
    networks:
      - myai_network


# --- Certbot for SSL Certificates ---
  certbot:
    image: certbot/certbot:latest
    container_name: myai_certbot
    volumes:
      - ./certbot/www:/usr/share/nginx/html
      - ./certbot/conf:/etc/letsencrypt
    networks:
      - myai_network

# --- Persistent Data Volumes ---
volumes:
  pg_data:

  qdrant_test_data:
